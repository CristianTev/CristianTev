Fishing Game

A fishing game where you control a fishing rod trying to catch as many fish as possible before time runs out. Fish swim beneath you while you move your rod left and right to hook them.

static void HandleInput()
    {
        while (Console.KeyAvailable)
        {
           var key = Console.ReadKey(true).Key;
           if (key == ConsoleKey.Escape) Environment.Exit(0);
        
           if ((key == ConsoleKey.Spacebar || key == ConsoleKey.DownArrow) && !rodExtending && !rodRetracting)
           rodExtending = true;
        
           if (key == ConsoleKey.LeftArrow)
            rodX--;
           if (key == ConsoleKey.RightArrow)
            rodX++;
        }
    }
